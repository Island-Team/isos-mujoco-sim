cmake_minimum_required(VERSION 3.22)
project(eos_mujoco_sim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Protobuf REQUIRED)
if(Protobuf_VERSION VERSION_LESS 3.12)
  message(FATAL_ERROR "Protobuf 3.12 or newer is required for proto3 optional fields")
endif()
if(Protobuf_VERSION VERSION_LESS 3.15)
  set(PROTO_OPTIONAL_ARGS --experimental_allow_proto3_optional)
endif()
find_package(gRPC CONFIG QUIET)
if(NOT gRPC_FOUND)
  # Ubuntu packages can provide pkg-config metadata without a CMake config.
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GRPC REQUIRED IMPORTED_TARGET grpc++)
  add_library(gRPC::grpc++ ALIAS PkgConfig::GRPC)
  find_program(GRPC_CPP_PLUGIN_EXECUTABLE NAMES grpc_cpp_plugin REQUIRED)
  add_executable(gRPC::grpc_cpp_plugin IMPORTED)
  set_target_properties(gRPC::grpc_cpp_plugin PROPERTIES
    IMPORTED_LOCATION "${GRPC_CPP_PLUGIN_EXECUTABLE}")
endif()
find_package(Threads REQUIRED)
find_package(JPEG REQUIRED)
find_package(X11 REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_library(EGL_LIBRARY EGL REQUIRED)
set(MUJOCO_ROOT "$ENV{MUJOCO_ROOT}" CACHE PATH "MuJoCo installation root")
if(NOT MUJOCO_ROOT OR NOT IS_ABSOLUTE "${MUJOCO_ROOT}")
  message(FATAL_ERROR "Set MUJOCO_ROOT to an absolute MuJoCo installation directory")
endif()
foreach(REQUIRED_FILE IN ITEMS include/mujoco/mujoco.h lib/libmujoco.so.3.3.1 bin/compile bin/simulate)
  if(NOT EXISTS "${MUJOCO_ROOT}/${REQUIRED_FILE}")
    message(FATAL_ERROR "MUJOCO_ROOT=${MUJOCO_ROOT} is missing ${REQUIRED_FILE}; select the installed MuJoCo directory")
  endif()
endforeach()
unset(MUJOCO_INCLUDE_DIR CACHE)
unset(MUJOCO_LIBRARY CACHE)
set(MUJOCO_INCLUDE_DIR "${MUJOCO_ROOT}/include")
set(MUJOCO_LIBRARY "${MUJOCO_ROOT}/lib/libmujoco.so")
add_library(mujoco SHARED IMPORTED)
set_target_properties(mujoco PROPERTIES IMPORTED_LOCATION "${MUJOCO_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${MUJOCO_INCLUDE_DIR}")
set(PROTO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/proto")
set(GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED_DIR}")
add_custom_command(OUTPUT "${GENERATED_DIR}/scene_devices.h"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/device.py" "${GENERATED_DIR}/scene_devices.h"
  DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tools/device.py" "${CMAKE_CURRENT_SOURCE_DIR}/tools/catalog.py" VERBATIM)
add_custom_target(scene_devices_generated DEPENDS "${GENERATED_DIR}/scene_devices.h")
foreach(NAME simulator benchmark)
  add_custom_command(OUTPUT "${GENERATED_DIR}/${NAME}.pb.cc" "${GENERATED_DIR}/${NAME}.pb.h" "${GENERATED_DIR}/${NAME}.grpc.pb.cc" "${GENERATED_DIR}/${NAME}.grpc.pb.h"
    COMMAND protobuf::protoc ${PROTO_OPTIONAL_ARGS} --proto_path=${PROTO_DIR} --cpp_out=${GENERATED_DIR} --grpc_out=${GENERATED_DIR} --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> "${PROTO_DIR}/${NAME}.proto"
    DEPENDS "${PROTO_DIR}/${NAME}.proto" VERBATIM)
  list(APPEND GENERATED_SOURCES "${GENERATED_DIR}/${NAME}.pb.cc" "${GENERATED_DIR}/${NAME}.grpc.pb.cc")
endforeach()
add_custom_target(proto_generated DEPENDS ${GENERATED_SOURCES})
add_subdirectory(src/mujoco)
add_subdirectory(src/isos)
add_subdirectory(src/plugin)
